# Vender v3 (S3) — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Orçamentos e Pedidos com etiquetas que dizem o estado e o próximo passo. O editor fica numa tela em blocos
com o total fixo no rodapé e a condição de pagamento num bloco só. Entram também três atalhos para orçar mais rápido
(grade, preço sugerido, Ctrl+S, opcionais em um clique e copiar especificação) e as pendências que vieram do S1 e do S2.

**Architecture:**
- **O que se adapta:** `Budgets.tsx`, `BudgetEditor.tsx` e `Orders.tsx`.
- **O que vai para funções puras testadas:**

  - `src/utils/condicaoDePagamento.ts`;
  - `src/utils/vendas/grade.ts`.
- **O que muda em funções que já existem:** `msgEntregue`, que passa a receber o saldo real.
- **O que é reusado:**
  - `etiquetaDoPedido` e `avisoDePrazo` (`src/utils/etiquetaDoPedido.ts`);
  - `resumirPagamentoDoPedido` (`src/utils/situacaoDoPedido.ts`);
  - `reescalarPlano` (`src/utils/planoDoPedido.ts`).
- **Banco:** sem SQL.

**Tech Stack:** React 18, TypeScript, Vite, Tailwind v4, Supabase. Testes com `node --test --experimental-strip-types`.

/**
 * Etiquetas que direcionam: dizem em que pé o orçamento/pedido está e qual é o próximo passo.
 * Spec do S3, seção 2. O "Enviado" usa a classificação da fila de retomada — os dois nunca discordam.
 */
import { filaDeFollowUp } from '../acompanhamentoComercial.ts';
import type { OrcamentoEnviado } from '../acompanhamentoComercial.ts';

export type TomDeVenda = 'neutro' | 'ambar' | 'rosa' | 'verde';
export type AcaoDoOrcamento = 'enviar' | 'retomar' | 'revalidar' | 'virar-pedido' | 'ver-pedido';

export interface EtiquetaDeOrcamento { texto: string; tom: TomDeVenda; acao: AcaoDoOrcamento; rotuloDaAcao: string }
export interface OrcamentoParaEtiquetar extends OrcamentoEnviado { generated_order_number?: string | null }

const ddmm = (iso: string) => `${iso.slice(8, 10)}/${iso.slice(5, 7)}`;
const somarDias = (iso: string, dias: number): string =>
    new Date(Date.parse(`${iso.slice(0, 10)}T00:00:00Z`) + dias * 86400000).toISOString().slice(0, 10);

const RETOMAR = { acao: 'retomar' as const, rotuloDaAcao: 'Retomar' };

export function etiquetaDoOrcamento(o: OrcamentoParaEtiquetar, hoje: string): EtiquetaDeOrcamento {
    const status = (o.status || '').trim();
    if (status === 'Converted') {
        return { texto: o.generated_order_number ? `Pedido #${o.generated_order_number}` : 'Pedido', tom: 'neutro', acao: 'ver-pedido', rotuloDaAcao: 'Ver pedido' };
    }
    if (status === 'Approved') return { texto: 'Aprovado', tom: 'verde', acao: 'virar-pedido', rotuloDaAcao: 'Virar pedido' };
    if (status === 'Sent') {
        const [item] = filaDeFollowUp([o], hoje);
        const vence = o.date && Number(o.validity_days) > 0 ? somarDias(o.date, Number(o.validity_days)) : '';
        const dias = item.diasParado;
        switch (item.urgencia) {
            case 'vencido': return { texto: `Vencido em ${ddmm(vence)}`, tom: 'rosa', acao: 'revalidar', rotuloDaAcao: 'Refazer validade' };
            case 'vence-hoje': return { texto: 'Vence hoje', tom: 'ambar', ...RETOMAR };
            case 'vencendo': return { texto: `Vence em ${ddmm(vence)}`, tom: 'ambar', ...RETOMAR };
            case 'parado': return { texto: dias === null ? 'Enviado' : `Parado há ${dias} dias`, tom: 'ambar', ...RETOMAR };
            default: return { texto: dias === null ? 'Enviado' : `Enviado há ${dias} ${dias === 1 ? 'dia' : 'dias'}`, tom: 'ambar', ...RETOMAR };
        }
    }
    return { texto: 'Rascunho', tom: 'neutro', acao: 'enviar', rotuloDaAcao: 'Enviar ao cliente' };
}

export type AcaoDoPedido = 'confirmar' | 'andamento' | 'ficha-de-entrega';

/** A ação à vista no cartão do pedido. `ultimoSetor` = nome do último setor da produção. */
export function acaoDoPedido(
    p: { status?: string | null; production_stage?: string | null },
    ultimoSetor: string | null,
): { acao: AcaoDoPedido; rotulo: string } {
    const status = (p.status || '').trim();
    const noUltimo = !!ultimoSetor && (p.production_stage || '').trim() === ultimoSetor.trim();
    if (status === 'Entregue' || noUltimo) return { acao: 'ficha-de-entrega', rotulo: 'Ficha de entrega' };
    if (status === 'Em Produção') return { acao: 'andamento', rotulo: 'Avisar andamento' };
    return { acao: 'confirmar', rotulo: 'Confirmar ao cliente' };
}

/** Classes da etiqueta, no formato pílula do estilo.ts (mesma altura, só muda a cor). */
export const CLASSE_DO_TOM_DE_VENDA: Record<TomDeVenda, string> = {
    neutro: 'bg-zinc-800/60 text-zinc-400 border-zinc-700/60',
    ambar: 'bg-amber-500/15 text-amber-400 border-amber-500/30',
    rosa: 'bg-rose-500/15 text-rose-400 border-rose-500/30',
    verde: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30',
};
```

  Os valores de status (`'Pendente'`, `'Em Produção'`, `'Entregue'`) são os do enum `OrderStatus`
  (`src/types/index.ts`); a função pura não importa o enum para continuar testável.

/**
 * Condição de pagamento num bloco só: a entrada % é a 1ª parcela do plano.
 *
 * É a mesma regra que o editor já usava ao gravar (a 1ª parcela do plano vira
 * down_payment_*), agora também na hora de editar — acabou o conflito entre o
 * campo "Entrada mínima %" e o plano.
 */
import { reescalarPlano } from './planoDoPedido.ts';
import type { PlanInstallment } from './paymentPlan.ts';

const centavos = (n: number) => Math.round(n * 100) / 100;

export function aplicarEntrada(o: {
    plano?: PlanInstallment[] | null; total: number; entradaPct: number;
    hoje: string; entrega: string; metodo?: string;
}): PlanInstallment[] {
    const pct = Math.min(99, Math.max(1, Math.round(Number(o.entradaPct) || 0)));
    const entrada = centavos(o.total * pct / 100);
    const resto = centavos(o.total - entrada);
    const atual = o.plano || [];
    if (atual.length >= 2) {
        const saldo = reescalarPlano(atual.slice(1), resto) as PlanInstallment[];
        return [{ ...atual[0], label: `Entrada (${pct}%)`, amount: entrada }, ...saldo];
    }
    const m = o.metodo || 'PIX';
    return [
        { label: `Entrada (${pct}%)`, amount: entrada, due: o.hoje, method: m },
        { label: 'Saldo na entrega', amount: resto, due: o.entrega, method: m },
    ];
}

/** O plano que o bloco mostra ao abrir. Orçamento antigo só com entrada % ganha o plano equivalente — sem gravar. */
export function planoParaMostrar(
    o: { payment_plan?: PlanInstallment[] | null; down_payment_percent?: number | null },
    total: number, hoje: string, entrega: string,
): PlanInstallment[] | null {
    if (o.payment_plan && o.payment_plan.length > 0) return o.payment_plan;
    const pct = Number(o.down_payment_percent) || 0;
    if (pct > 0 && pct < 100 && total > 0) return aplicarEntrada({ total, entradaPct: pct, hoje, entrega });
    return null;
}

/** Entrada % do plano = 1ª parcela ÷ total (a mesma conta que handleSave grava em down_payment_percent). */
export const entradaPctDoPlano = (plano: PlanInstallment[] | null | undefined, total: number): number =>
    plano && plano.length > 0 && total > 0 ? Math.round(((Number(plano[0].amount) || 0) / total) * 100) : 0;

/**
 * A grade de tamanhos digitada numa linha só: "P 10, M 20, G 15".
 * Cada tamanho continua sendo uma linha do orçamento — isto só poupa a digitação.
 */
export type ResultadoDaGrade =
    | { ok: true; tamanhos: { size: string; quantity: number }[] }
    | { ok: false; erro: string };

// Letras: "P10", "P 10", "P:10", "P=10". Números exigem separador, senão "364" seria 36 × 4 ou 3 × 64.
const LETRAS = /^([A-Za-z]{1,4})\s*[:=]?\s*(\d+)$/;
const NUMERO = /^(\d{1,3})\s*(?:[:=]|\s)\s*(\d+)$/;

export function lerGrade(texto: string): ResultadoDaGrade {
    const partes = (texto || '').split(/[,;\n]+/).map(p => p.trim()).filter(Boolean);
    if (partes.length === 0) return { ok: false, erro: 'Digite a grade, ex.: P 10, M 20, G 15' };
    const tamanhos: { size: string; quantity: number }[] = [];
    for (const parte of partes) {
        const m = parte.match(LETRAS) || parte.match(NUMERO);
        if (!m) return { ok: false, erro: `Não entendi "${parte}"` };
        const size = m[1].toUpperCase();
        const quantity = Number(m[2]);
        if (tamanhos.some(t => t.size === size)) return { ok: false, erro: `Tamanho repetido: ${size}` };
        if (quantity <= 0) return { ok: false, erro: `Quantidade zero em ${size}` };
        tamanhos.push({ size, quantity });
    }
    return { ok: true, tamanhos };
}
```

/**
 * Especificação da linha do orçamento: o resumo curto e a cópia de uma linha para outras.
 * Spec do S3, seção 5B. Os campos são os do BudgetItem de hoje — nada novo no banco.
 */
export interface ItemComEspecificacao {
    sleeve_type?: string; gender?: string; fit?: string; has_band?: boolean; prod_notes?: string;
    fabric_insumo_id?: string; fabric_name?: string; fabric: string; color: string;
    quantity: number; unit_price: number; total: number;
}

export type GrupoCopiavel = 'opcionais' | 'detalhes' | 'tecido' | 'preco';

const CAMPOS: Record<GrupoCopiavel, (keyof ItemComEspecificacao)[]> = {
    opcionais: ['sleeve_type', 'gender', 'fit', 'has_band'],
    detalhes: ['prod_notes'],
    // O insumo carrega a faixa de preço da cor: tecido e cor só fazem sentido juntos.
    tecido: ['fabric_insumo_id', 'fabric_name', 'fabric', 'color'],
    preco: ['unit_price'],
};

const centavos = (n: number) => Math.round(n * 100) / 100;

/** Copia da origem só os grupos pedidos. "Não definido" na origem também é copiado (limpa o destino). */
export function copiarEspecificacao<T extends ItemComEspecificacao>(origem: T, destino: T, grupos: GrupoCopiavel[]): T {
    const novo = { ...destino };
    for (const g of grupos) for (const campo of CAMPOS[g]) (novo as any)[campo] = (origem as any)[campo];
    if (grupos.includes('preco')) novo.total = centavos((Number(novo.quantity) || 0) * (Number(novo.unit_price) || 0));
    return novo;
}

const ROTULO: Record<string, string> = {
    manga_curta: 'Manga curta', manga_longa: 'Manga longa', sem_manga: 'Sem manga', outro: 'Manga: outro',
    masculino: 'Masc', feminino: 'Fem', infantil: 'Infantil', unissex: 'Unissex',
    slim: 'Slim', regular: 'Regular', oversized: 'Oversized', babylook: 'Babylook', plus_size: 'Plus size',
};

export function resumoDaEspecificacao(item: ItemComEspecificacao): string {
    const partes = [item.sleeve_type, item.gender, item.fit]
        .filter((v): v is string => !!v)
        .map(v => ROTULO[v] || v);
    if (item.has_band) partes.push('Faixa');
    return partes.length ? partes.join(' · ') : 'sem especificação';
}
