```ts
/**
 * "Receber compra": a lista de compras vira a janela de entrada de uma vez só.
 * Spec do S4, §5.3. Uma entrada por insumo, pelo mesmo caminho da entrada de hoje.
 */
import type { PurchaseLine } from '../purchaseList.ts';
import { parseBRL } from '../currency.ts';

export interface LinhaDeRecebimento {
    insumo_id: string;
    nome: string;
    unidade: string;
    marcado: boolean;
    /** Texto do campo, como a pessoa digita ("12,5"). */
    quantidade: string;
    custo: string;
}

const texto = (n: number) => String(Math.round(n * 1000) / 1000).replace('.', ',');
const centavos = (n: number) => Math.round(n * 100) / 100;

export function linhasParaReceber(lines: PurchaseLine[]): LinhaDeRecebimento[] {
    return lines
        .filter(l => l.is_shortage && l.to_buy > 0)
        .map(l => ({
            insumo_id: l.insumo_id,
            nome: l.insumo_name,
            unidade: l.unit,
            marcado: true,
            quantidade: texto(l.to_buy),
            // Custo zerado no cadastro não vira "0": vem vazio, para a pessoa preencher.
            custo: l.unit_cost > 0 ? texto(l.unit_cost) : '',
        }));
}

export function validarRecebimento(linhas: LinhaDeRecebimento[]) {
    const erros: Record<string, string> = {};
    let marcadas = 0;
    for (const l of linhas) {
        if (!l.marcado) continue;
        marcadas++;
        if (!(parseBRL(l.quantidade) > 0)) erros[l.insumo_id] = 'Informe a quantidade';
        else if (!(parseBRL(l.custo) > 0)) erros[l.insumo_id] = 'Informe o custo';
    }
    return { erros, podeGravar: marcadas > 0 && Object.keys(erros).length === 0, marcadas };
}

export function totalDoRecebimento(linhas: LinhaDeRecebimento[]): number {
    return centavos(linhas.filter(l => l.marcado)
        .reduce((t, l) => t + parseBRL(l.quantidade) * parseBRL(l.custo), 0));
}

export function paraLancar(linhas: LinhaDeRecebimento[]) {
    return linhas.filter(l => l.marcado).map(l => ({
        insumo_id: l.insumo_id,
        quantity: parseBRL(l.quantidade),
        unit_cost: parseBRL(l.custo),
    }));
}
```

```ts
/** Aviso do Cronograma (spec S4 §4.3): só existe quando falta insumo; leitura falha ou carregando = null. */
export function avisoDeCompras(faltas: number | null): string | null {
    if (faltas === null || faltas <= 0) return null;
    return faltas === 1
        ? 'Falta 1 insumo para os pedidos em aberto'
        : `Faltam ${faltas} insumos para os pedidos em aberto`;
}
```

```ts
/** "Estoque crítico" como filtro (spec S4 §6.2). Mesmo critério do indicador de hoje. */
export const ehCritico = (i: { current_stock: number; min_stock: number }) =>
    (i.min_stock || 0) > 0 && i.current_stock <= i.min_stock;

export function filtrarCriticos<T extends { name: string; category?: string | null; current_stock: number; min_stock: number }>(
    itens: T[], busca: string, categoria: string, semCategoria: string,
): T[] {
    const b = busca.trim().toLowerCase();
    return itens.filter(i =>
        ehCritico(i)
        && (!b || i.name.toLowerCase().includes(b))
        && (!categoria || ((i.category || '').trim() || semCategoria) === categoria));
}
```
